// Set to 150 minutes (150 * 60 = 9000 seconds) let timeRemainingSeconds = 150 * 60; let timerInterval = null; let isTimerPaused = false; function updateTimerDisplay() { const hrs = String(Math.floor(timeRemainingSeconds / 3600)).padStart(2, '0'); const mins = String(Math.floor((timeRemainingSeconds % 3600) / 60)).padStart(2, '0'); const secs = String(timeRemainingSeconds % 60).padStart(2, '0'); document.getElementById("timerDisplay").textContent = `${hrs}:${mins}:${secs}`; } function startTimer() { updateTimerDisplay(); timerInterval = setInterval(() => { if (!isTimerPaused) { if (timeRemainingSeconds > 0) { timeRemainingSeconds--; updateTimerDisplay(); } else { clearInterval(timerInterval); alert("Time is up!"); submitExam(); } } }, 1000); } // Pause / Resume Toggle Handler document.getElementById("btnPauseTimer").onclick = () => { if (isSubmitted) return; isTimerPaused = !isTimerPaused; const btn = document.getElementById("btnPauseTimer"); if (isTimerPaused) { btn.textContent = "▶ Resume"; btn.className = "bg-emerald-500 hover:bg-emerald-600 text-white font-bold text-xs px-2.5 py-1.5 rounded shadow transition"; } else { btn.textContent = "⏸ Pause"; btn.className = "bg-amber-500 hover:bg-amber-600 text-slate-950 font-bold text-xs px-2.5 py-1.5 rounded shadow transition"; } };
